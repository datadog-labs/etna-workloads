extern crate std;
#[attr = PreludeImport]
use std::prelude::rust_2024::*;
fn main() {
    let x = 1;
    let _t =
        match x {
            _ => {
                'a: loop {
                    {
                        ::std::io::_print(format_arguments::from_str("a long enough line to force wrapping of the printed output, yes\n"));
                    };
                    {
                        ::std::io::_print(format_arguments::from_str("another long enough line to force wrapping of the printed output\n"));
                    };
                    break 'a;
                }
            }
        };
}
status 0

[exit 0]
